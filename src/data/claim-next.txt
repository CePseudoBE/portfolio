async fn claim_next(&self) -> Result<Option<ExportJob>, DomainError> {
    // One statement: the sub-select locks a single candidate and SKIP LOCKED
    // steps over rows another worker is taking, so N workers claim N jobs.
    let row = sqlx::query(&format!(
        "UPDATE export_jobs SET state = 'running', claimed_at = now(), \
                                attempts = attempts + 1 \
         WHERE id = ( \
             SELECT id FROM export_jobs \
             WHERE state = 'pending' \
             ORDER BY created_at \
             FOR UPDATE SKIP LOCKED \
             LIMIT 1 \
         ) \
         RETURNING {JOB_COLUMNS}"
    ))
    .fetch_optional(&self.pool)
